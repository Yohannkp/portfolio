### Core Web Vitals Monitoring
```javascript
// Monitoring performance en temps réel
const vitalsReporter = (metric) => {
  // Envoyer les métriques à votre service d'analytics
  console.log(`${metric.name}: ${metric.value}`);
  
  // Seuils d'alerte
  if (metric.name === 'LCP' && metric.value > 2500) {
    console.warn('LCP trop élevé:', metric.value);
  }
  
  if (metric.name === 'CLS' && metric.value > 0.1) {
    console.warn('CLS problématique:', metric.value);
  }
};

// Utilisation avec web-vitals library
import {getLCP, getFID, getCLS} from 'web-vitals';
getLCP(vitalsReporter);
getFID(vitalsReporter);  
getCLS(vitalsReporter);
```
